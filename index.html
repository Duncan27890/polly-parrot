    // Firebase configuration and state

    const firebaseConfig = JSON.parse(__firebase_config);

    const app = initializeApp(firebaseConfig);

    const auth = getAuth(app);

    const db = getFirestore(app);

    const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';


    let user = null;


    // --- FIREBASE AUTHENTICATION ---

    const initAuth = async () => {

        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

            await signInWithCustomToken(auth, __initial_auth_token);

        } else {

            await signInAnonymously(auth);

        }

    };


    onAuthStateChanged(auth, (u) => {

        user = u;

    async function saveScore(score) {

        if (!user) return;

        try {

            // Rule 1: Correct path structure

            const scoreRef = doc(db, 'artifacts', appId, 'public', 'data', 'scores', user.uid);

            await setDoc(scoreRef, {

                score: score,

                userId: user.uid,

                timestamp: Date.now()

            });

        } catch (e) {

            console.error("Error saving score", e);

        }

    }


    async function fetchLeaderboard() {

        if (!user) return [];

        try {

            const colRef = collection(db, 'artifacts', appId, 'public', 'data', 'scores');

            const snapshot = await getDocs(colRef);

            let scores = [];

            snapshot.forEach(doc => {

                scores.push(doc.data());

            });

            // Rule 2: Sort in-memory to avoid index requirements

            return scores.sort((a, b) => b.score - a.score).slice(0, 10);

        } catch (e) {

            console.error("Error fetching leaderboard", e);

            return [];

        }

    }

    async function initAudio() {

        if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();

        if (audioCtx.state === 'suspended') await audioCtx.resume();

    }


    function playFlapSound() {

        if (!audioCtx) return;

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = 'triangle';

        osc.frequency.setValueAtTime(150, audioCtx.currentTime);

        osc.frequency.exponentialRampToValueAtTime(400, audioCtx.currentTime + 0.1);

        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);

        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.1);

        osc.connect(gain); gain.connect(audioCtx.destination);

        osc.start(); osc.stop(audioCtx.currentTime + 0.1);

    }


    function playOinkSound() {

        if (!audioCtx) return;

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = 'sawtooth';

        osc.frequency.setValueAtTime(80, audioCtx.currentTime);

        osc.frequency.linearRampToValueAtTime(40, audioCtx.currentTime + 0.2);

        gain.gain.setValueAtTime(0.05, audioCtx.currentTime);

        gain.gain.linearRampToValueAtTime(0, audioCtx.currentTime + 0.2);

        osc.connect(gain); gain.connect(audioCtx.destination);

        osc.start(); osc.stop(audioCtx.currentTime + 0.2);

    }


    function playScoreSound() {

        if (!audioCtx) return;

        const osc = audioCtx.createOscillator();

        const gain = audioCtx.createGain();

        osc.type = 'sine';

        osc.frequency.setValueAtTime(600, audioCtx.currentTime);

        osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.1);

        gain.gain.setValueAtTime(0.1, audioCtx.currentTime);

        gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.2);

        osc.connect(gain); gain.connect(audioCtx.destination);

        osc.start(); osc.stop(audioCtx.currentTime + 0.2);

    }


    async function fetchVoiceLine(text) {

        const apiKey = ""; // Runtime provided

        const payload = {

            contents: [{ parts: [{ text: `Say in a high-pitched, cheerful British child's voice: ${text}` }] }],

            generationConfig: {

                responseModalities: ["AUDIO"],

                speechConfig: { voiceConfig: { prebuiltVoiceConfig: { voiceName: "Kore" } } }

            },

            model: "gemini-2.5-flash-preview-tts"

        };

        const url = `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-tts:generateContent?key=${apiKey}`;

        try {

            const response = await fetch(url, { method: 'POST', body: JSON.stringify(payload) });

            const result = await response.json();

            const audioData = result.candidates[0].content.parts[0].inlineData.data;

            return await decodePCM(audioData);

        } catch (err) { return null; }

    }


    async function decodePCM(base64Data) {

        const bytes = Uint8Array.from(atob(base64Data), c => c.charCodeAt(0));

        const buffer = audioCtx.createBuffer(1, bytes.length / 2, 24000);

        const channelData = buffer.getChannelData(0);

        const dataView = new DataView(bytes.buffer);

        for (let i = 0, j = 0; i < bytes.length; i += 2, j++) {

            channelData[j] = dataView.getInt16(i, true) / 32768;

        }

        return buffer;

    }


    function playAudioBuffer(buffer) {

        if (!buffer || !audioCtx) return;

        const source = audioCtx.createBufferSource();

        source.buffer = buffer;

        source.connect(audioCtx.destination);

        source.start();

    }

    class Bird {

        constructor() { this.reset(); }

        reset() {

            this.x = 100; this.y = canvas.height / 2;

            this.velocity = 0; this.radius = 18; this.rotation = 0;

        }

        update() {

            this.velocity += GRAVITY;

            this.y += this.velocity;

            this.rotation = Math.min(Math.PI / 4, Math.max(-Math.PI / 4, this.velocity * 0.08));

            if (this.y + this.radius > canvas.height - 40) endGame();

            if (this.y - this.radius < 0) { this.y = this.radius; this.velocity = 0; }

        }

        flap() { this.velocity = FLAP_STRENGTH; playFlapSound(); }

        draw() {

            ctx.save();

            ctx.translate(this.x, this.y); ctx.rotate(this.rotation);

            ctx.fillStyle = '#4caf50'; ctx.strokeStyle = '#2e7d32'; ctx.lineWidth = 3;

            ctx.beginPath(); ctx.ellipse(0, 0, 24, 20, 0, 0, Math.PI * 2); ctx.fill(); ctx.stroke();

            ctx.fillStyle = '#ff5252'; ctx.beginPath(); ctx.moveTo(18, -6); ctx.quadraticCurveTo(35, 0, 18, 12); ctx.closePath(); ctx.fill(); ctx.stroke();

            ctx.fillStyle = 'white'; ctx.beginPath(); ctx.arc(10, -8, 7, 0, Math.PI * 2); ctx.fill(); ctx.stroke();

            ctx.fillStyle = 'black'; ctx.beginPath(); ctx.arc(12, -8, 3, 0, Math.PI * 2); ctx.fill();

            ctx.fillStyle = '#81c784'; ctx.beginPath(); ctx.ellipse(-10, 4, 14, 9, Math.PI/6, 0, Math.PI * 2); ctx.fill(); ctx.stroke();

            ctx.restore();

        }

    }


    class Pipe {

        constructor(x) {

            this.x = x;

            this.topHeight = Math.random() * (canvas.height - PIPE_GAP - 200) + 80;

            this.bottomY = this.topHeight + PIPE_GAP;

            this.passed = false;

            this.color = ['#ff8a80', '#80d8ff', '#ccff90', '#ffff8d'][Math.floor(Math.random() * 4)];

        }

        update() { this.x -= PIPE_SPEED; }

        draw() {

            ctx.lineWidth = 5; ctx.strokeStyle = '#555'; ctx.fillStyle = this.color;

            ctx.beginPath(); ctx.roundRect(this.x, 0, PIPE_WIDTH, this.topHeight, [0, 0, 15, 15]); ctx.fill(); ctx.stroke();

            ctx.beginPath(); ctx.roundRect(this.x, this.bottomY, PIPE_WIDTH, canvas.height - this.bottomY, [15, 15, 0, 0]); ctx.fill(); ctx.stroke();

        }

    }

    function endGame() {

        if (gameState === 'GAMEOVER') return;

        gameState = 'GAMEOVER';

        playAudioBuffer(sounds.ohDear);

        playOinkSound();

        finalScoreTxt.innerText = score;

        if (score > highScore) {

            highScore = score;

            localStorage.setItem('pollyHighScore', highScore);

            updateHighScoreUI();

        }

        saveScore(score); 

        gameOverScreen.classList.add('visible');

    }


    function handleInput() { if (gameState === 'PLAYING') bird.flap(); }

    window.addEventListener('mousedown', (e) => { if (e.target.tagName !== 'BUTTON') handleInput(); });

    window.addEventListener('touchstart', (e) => { if (e.target.tagName !== 'BUTTON') { e.preventDefault(); handleInput(); } }, { passive: false });

    window.addEventListener('keydown', (e) => { if (e.code === 'Space') handleInput(); });


    function drawBackground() {

        backgroundX = (backgroundX - 0.5) % (canvas.width * 2);

        // Sun

        ctx.fillStyle = '#fff176'; ctx.beginPath(); ctx.arc(canvas.width - 70, 80, 40, 0, Math.PI * 2); ctx.fill();

        // Hills

        ctx.fillStyle = '#a2e32e'; ctx.beginPath(); ctx.arc(canvas.width / 2 + (backgroundX * 0.3), canvas.height + 150, 450, 0, Math.PI * 2); ctx.fill();

        // Ground

        ctx.fillStyle = '#d4e157'; ctx.fillRect(0, canvas.height - 40, canvas.width, 40);

    }


    function loop() {

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        drawBackground();

        if (gameState === 'PLAYING') {

            frameCount++;

            if (frameCount % PIPE_SPAWN_RATE === 0) pipes.push(new Pipe(canvas.width));

            bird.update();

            for (let i = pipes.length - 1; i >= 0; i--) {

                const p = pipes[i]; p.update(); p.draw();

                if (bird.x + bird.radius > p.x && bird.x - bird.radius < p.x + PIPE_WIDTH && (bird.y - bird.radius < p.topHeight || bird.y + bird.radius > p.bottomY)) endGame();

                if (!p.passed && bird.x > p.x + PIPE_WIDTH) { p.passed = true; score++; scoreDisplay.innerText = score; playScoreSound(); }

                if (p.x + PIPE_WIDTH < 0) pipes.splice(i, 1);

            }

            bird.draw();

        } else { if (bird) { bird.y = canvas.height/2 + Math.sin(Date.now() / 300) * 15; bird.draw(); } else { bird = new Bird(); } }
